function main(): void {
    let a: int;
    let b: int;
    let max: int;
    let es_mayor: char;

    a = 5;
    b = 10;

    // 1. Prueba básica (Condición Falsa)
    // 'a' no es mayor que 'b', por lo que debe devolver 'b'
    max = a > b ? a : b;
    log 'M', 'a', 'x', '1', ':', max, '\n';

    // 2. Prueba básica (Condición Verdadera)
    a = 15;
    // Ahora 'a' sí es mayor que 'b', debe devolver 'a'
    max = a > b ? a : b;
    log 'M', 'a', 'x', '2', ':', max, '\n';

    // 3. Prueba con otros tipos (char) y literales fijos
    es_mayor = a > 2 ? 'Y' : 'N';
    log 'A', '>', '2', '?', ' ', es_mayor, '\n';
}
