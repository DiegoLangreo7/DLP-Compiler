// ==========================================
// VARIABLES GLOBALES (Tipos simples, Arrays y Registros)
// ==========================================

let maxSimulations: int;
let baseLatency: number;
let is5G_NSA: int;

let networkStats: [
    let packetsSent: int;
    let packetsDropped: int;
    let averageJitter: number;
];

let latencyHistory: [10] number;

// ==========================================
// FUNCIONES DE SOPORTE
// ==========================================

function calculateLatency(distance: number, signalStrength: int): number {
    // DECLARACIONES AL PRINCIPIO
    let latency: number;
    let interference: number;
    let base: number;

    // SENTENCIAS LUEGO
    base = 10.0;

    // Promoción de tipos y casting
    interference = ((100 - signalStrength) as number) * 0.5;

    if (is5G_NSA == 1) {
        // Simulación 5G NSA
        latency = baseLatency + (distance * 0.1) + interference;
    } else {
        // Simulación 4G LTE (doble latencia base)
        latency = (baseLatency * 2.0) + (distance * 0.2) + interference;
    }

    return latency;
}

function simulatePing(index: int, distance: number, strength: int): void {
    let currentLat: number;
    let dropChance: int;

    // Llamada a función como expresión
    currentLat = calculateLatency(distance, strength);

    // Operador módulo
    dropChance = strength % 10;

    networkStats.packetsSent = networkStats.packetsSent + 1;

    // Condicionales lógicos complejos
    if (dropChance < 2 && strength < 50) {
        networkStats.packetsDropped = networkStats.packetsDropped + 1;
        latencyHistory[index] = 999.9;
        log 'D', 'r', 'o', 'p', 'p', 'e', 'd', '\n';
    } else {
        latencyHistory[index] = currentLat;
        log 'P', 'i', 'n', 'g', ':', currentLat, '\n';
    }
}

// ==========================================
// FUNCIÓN PRINCIPAL
// ==========================================

function main(): void {
    // 1. ZONA DE DECLARACIONES
    let i, tempStr, validPackets: int;
    let totalLat: number;

    // 2. ZONA DE INICIALIZACIONES (Sentencias)
    maxSimulations = 10;
    baseLatency = 15.5;
    is5G_NSA = 1; // 1 = 5G NSA, 0 = 4G LTE

    networkStats.packetsSent = 0;
    networkStats.packetsDropped = 0;
    networkStats.averageJitter = 0.0;

    log 'T', 'e', 'l', 'e', 'm', 'e', 't', 'r', 'y', '\n';

    i = 0;
    totalLat = 0.0;
    tempStr = 85; // Fuerza de señal inicial

    // 3. BUCLE PRINCIPAL DE SIMULACIÓN
    while (i < maxSimulations) {
        // Llamada a procedimiento (como sentencia)
        simulatePing(i, 50.0, tempStr);

        if (latencyHistory[i] != 999.9) {
            totalLat = totalLat + latencyHistory[i];
        }

        // Empeoramos la señal progresivamente
        tempStr = tempStr - 12;

        if (tempStr < 0) {
            tempStr = 100; // Reset si la señal cae demasiado
        }

        i = i + 1;
    }

    // 4. CÁLCULO FINAL DE ESTADÍSTICAS
    validPackets = networkStats.packetsSent - networkStats.packetsDropped;

    if (validPackets > 0) {
         networkStats.averageJitter = totalLat / (validPackets as number);
    }

    log '\n', '-', '-', '-', '\n';
    log 'S', 'e', 'n', 't', ':', networkStats.packetsSent, '\n';
    log 'D', 'r', 'o', 'p', 's', ':', networkStats.packetsDropped, '\n';
    log 'A', 'v', 'g', 'L', 'a', 't', ':', networkStats.averageJitter, '\n';
}
