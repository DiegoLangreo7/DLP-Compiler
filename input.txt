function testReturnVacio(cortar: int): void {
    log 'I', 'N', 'I', 'C', 'I', 'O', '\n';

    if (cortar == 1) {
        log 'A', 'D', 'I', 'O', 'S', '\n';
        // ¡La nueva estrella del compilador!
        return;
    }

    // Si 'cortar' era 1, el programa NUNCA debería llegar a imprimir esto:
    log 'F', 'I', 'N', 'A', 'L', '\n';
}

function main(): void {
    log 'P', 'R', 'U', 'E', 'B', 'A', ' ', '1', '\n';
    // Le pasamos 1 para que haga el return temprano
    testReturnVacio(1);

    log '\n', 'P', 'R', 'U', 'E', 'B', 'A', ' ', '2', '\n';
    // Le pasamos 0 para que ignore el return y llegue al final
    testReturnVacio(0);
}
