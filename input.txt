function main(): void {
    let i: int;
    i = 0;

    log 'I', 'N', 'I', 'C', 'I', 'O', '\n';

    while (i < 10) {
        i = i + 1;

        if (i == 3) {
            // Saltamos la impresión del 3
            continue;
        }

        if (i == 6) {
            // Abortamos el bucle al llegar a 6
            break;
        }

        log i, '\n';
    }

    log 'F', 'I', 'N', '\n';
}
