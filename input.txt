// Función para calcular potencias (prueba paso de floats y bucles)
function power(base: number, exp: int): number {
    let result: number;
    let i: int;
    result = 1.0;
    i = 0;
    while (i < exp) {
        result = result * base;
        i = i + 1;
    }
    return result;
}

// Función para comprobar si un número es primo (prueba booleanos int, modulos y retornos tempranos simulados)
function isPrime(n: int): int {
    let i: int;
    let is_prime: int;
    is_prime = 1;

    if (n <= 1)
        is_prime = 0;
    else {
        i = 2;
        while (i * i <= n && is_prime == 1) {
            if (n % i == 0)
                is_prime = 0;
            i = i + 1;
        }
    }
    return is_prime;
}

// Procedimiento para imprimir un separador decorativo
function printSeparator(c: char, length: int): void {
    let i: int;
    i = 0;
    while (i < length) {
        log c;
        i = i + 1;
    }
    log '\n';
}
