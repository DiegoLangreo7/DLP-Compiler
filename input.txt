function main(): void {
    let i: int;
    let suma: int;

    suma = 0;

    log 'I', 'N', 'I', 'C', 'I', 'O', '\n';

    // Prueba de fuego del bucle for
    for (i = 0; i < 5; i = i + 1) {
        log 'i', ':', i, '\n';
        suma = suma + i;
    }

    log 'S', 'U', 'M', 'A', ':', suma, '\n';
    log 'F', 'I', 'N', '\n';
}
