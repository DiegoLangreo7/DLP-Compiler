// --- NIVEL 0: Ámbito Global ---
let a: int;

function testScopes(a: int): void {
    let flag: int;

    // --- NIVEL 1: Ámbito de Parámetros ---
    log 'P', 'A', 'R', 'A', 'M', ':', a, '\n';

    // --- NIVEL 2: Ámbito Local de Función ---
    // (Hacemos shadowing al parámetro)
    let a: int;
    a = 3;
    log 'F', 'U', 'N', 'C', ':', a, '\n';

    if (a == 3) {
        // --- NIVEL 3: Ámbito de Bloque (IF) ---
        // (Hacemos shadowing a la local de la función)
        let a: int;
        a = 4;
        log 'B', 'L', 'O', 'Q', 'U', 'E', ':', a, '\n';

        flag = 1;
        while (flag == 1) {
            // --- NIVEL 4: Ámbito de Bloque (WHILE) ---
            // (Hacemos shadowing a la del IF)
            let a: int;
            a = 5;
            log 'W', 'H', 'I', 'L', 'E', ':', a, '\n';

            // Apagamos el flag para salir de forma natural
            flag = 0;
        }
    }

    // Al salir del IF, la pila hace 'pop' y volvemos al Nivel 2 (debería valer 3)
    log 'V', 'U', 'E', 'L', 'T', 'A', ':', a, '\n';
}

function main(): void {
    a = 1;
    log 'G', 'L', 'O', 'B', 'A', 'L', ':', a, '\n';

    // Llamamos a la función pasando un 2 al parámetro
    testScopes(2);

    // Tras toda la locura, la global sigue intacta (debería valer 1)
    log 'F', 'I', 'N', 'A', 'L', ':', a, '\n';
}
