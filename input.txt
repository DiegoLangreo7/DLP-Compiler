function main(): void {
    let x: int;

    log 'P', 'R', 'U', 'E', 'B', 'A', 'S', '\n';

    // 1. Suma
    x = 10;
    x += 5;
    log '+', '=', ' ', x, '\n'; // Esperado: 15

    // 2. Resta
    x = 10;
    x -= 3;
    log '-', '=', ' ', x, '\n'; // Esperado: 7

    // 3. Multiplicación
    x = 10;
    x *= 2;
    log '*', '=', ' ', x, '\n'; // Esperado: 20

    // 4. División
    x = 10;
    x /= 2;
    log '/', '=', ' ', x, '\n'; // Esperado: 5

    // 5. Módulo (Resto)
    x = 10;
    x %= 3;
    log '%', '=', ' ', x, '\n'; // Esperado: 1

    // 6. OR a nivel de bits / Lógico
    // 1 (binario 01) | 2 (binario 10) = 3 (binario 11)
    x = 1;
    x |= 2;
    log '|', '=', ' ', x, '\n'; // Esperado: 3 o 1 (dependiendo de si tu '|' es bit a bit o puramente lógico)

    // 7. AND a nivel de bits / Lógico
    // 3 (binario 11) & 2 (binario 10) = 2 (binario 10)
    x = 3;
    x &= 2;
    log '&', '=', ' ', x, '\n'; // Esperado: 2 o 1 (dependiendo de tu implementación lógica)
}
