function main(): void {
    let dia: int;

    log 'P', 'R', 'U', 'E', 'B', 'A', ' ', '1', '\n';
    dia = 2;

    switch (dia) {
        case 1:
            log 'L', 'U', 'N', 'E', 'S', '\n';

        case 2:
            log 'M', 'A', 'R', 'T', 'E', 'S', '\n';

        case 3:
            log 'M', 'I', 'E', 'R', 'C', 'O', 'L', 'E', 'S', '\n';

        default:
            log 'O', 'T', 'R', 'O', '\n';
    }

    log '\n', 'P', 'R', 'U', 'E', 'B', 'A', ' ', '2', '\n';
    dia = 99;

    switch (dia) {
        case 1:
            log 'L', 'U', 'N', 'E', 'S', '\n';

        default:
            log 'O', 'T', 'R', 'O', '\n';
    }
}
